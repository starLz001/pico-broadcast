private bool isRunning=false; // 在调用 StartGetImageData 后，将状态置为 true
static bool  reopen = false;

private void Update()
{
    if (reopen)
    {
        // 在主线程里执行重新打开功能
        reopen = false;
        PXR_Enterprise.StartGetImageDatafor4U(Mode, (int)frame.width, (int)frame.height);  
    }
}

private void OnApplicationPause(bool pauseStatus)
{
    if (isRunning)
    {
        // 在已启动获取数据的功能后
        if (pauseStatus)
        {
            // 如果灭屏，执行 close 功能，释放相机
            PXR_Enterprise.CloseCamerafor4U();
        }
        else
        {
            // 再次灭屏或亮屏，重新打开相机
            PXR_Enterprise.OpenCameraAsyncfor4U(ret =>
            {
                Debug.Log($$"{tag}  OpenCameraAsync ret=  {ret}");
                // 修改重新打开的标志位
                reopen = ret;
            });
        }
    }
    
}
