太空竞技场
========================================
太空竞技场是展示玩家在 VR 世界中进行社交互动的 Demo，带你快速体验 PICO SDK 提供的“账号&好友” 服务、“社交互动”服务和“房间&匹配”服务。具体而言，玩家可以在 Demo 中体验创建房间、邀请他人进入房间、加入他人房间、在房间内发起好友申请等功能。
*预览视频
你可以通过以下视频快速了解该 Demo 的主要功能。
 
*环境要求
要体验本 Demo，你需要配置以下环境：
*虚幻引擎 4.27.2 版本
*Visual Studio 2019 或 2022 版本
*SDK 2.1.4 版本及以上
要了解如何在 Unreal Engine 4.27.2 中配置安卓环境，请参考配置安卓开发环境。
*设备支持
PICO Neo3、PICO 4。

*实现 Demo
本部分介绍自行实现 Demo 中社交互动能力的步骤。
*第一步：获取 Demo 项目
*前往 GitHub 下载社交互动 Demo 的项目文件夹到本地。
*解压该 Demo 压缩包到 Unreal Projects 目录下。
*将下载的 2.1.4 及以上版本的 SDK plugins 包移动到 Demo 文件夹内。
* 
*右键点击 PlatformSquare.uproject 文件，点击 Generate Visual Studio project files，生成 PlatformSquare.sln 文件。
* 
*双击 PlatformSquare.sln 文件，该文件将通过 Visual Studio 打开。
*在 Visual Studio 界面中，右键点击 PlatformSquare，在下拉选项中，点击 生成（Build）。
* 
*项目生成成功后，你可以双击 PlatformSquare.uproject 打开 UE 工程查看 Demo。
*第二步：创建应用
要使用 PICO 的平台服务能力，你需要前往 PICO 开发者平台创建应用。要了解更多信息，参考创建应用。
*第三步：开通平台服务
本 Demo 使用了 PICO 平台服务中的匹配服务和 Destinations 功能。因此，你需要在创建的应用中开通匹配服务并创建一个 Destination。
*登录 PICO 开发者平台。
*在 我的应用 页面，选择在第二步中创建的应用，进入应用概览页。
*在左侧导航栏，点击 平台服务 > 匹配服务 > 启动服务。
* 
*在左侧导航栏，点击 平台服务 > Destinations。为应用创建一个 Destination。配置项按需填写即可。
*第四步：为项目添加 App ID
*在 PICO 开发者平台上，前往先前所创建应用的 API 页面，获取该应用的 APP ID。 
* 
*在 Unreal 编辑器中打开 Demo 项目。
*前往 编辑（Edit） > 项目设置（Proejct Settings） > 插件（Plugins） > OnlinePico Settings，填写 AppID。
* 
*第五步：在项目配置文件中添加 Destination
在 Demo 项目文件夹中，打开 Config > DefaultEngine.ini 文件，填写你在开发者平台创建的 Destination 的名称。
* 
现在，你已经在 Demo 中实现了 PICO 平台服务的能力。你可以将 Demo 项目打包安装至 PICO 设备上开始体验。

void APicoSquarePlayerState::Login()
{
    PlatformLog(FString(TEXT("PicoSquarePlayerState::Login!")));
    LoginDelegate.Clear();
    LoginDelegate.BindDynamic(this, &APicoSquarePlayerState::OnLoginComplete);
    UOnlinePicoUserFunction::GetLoginUser(this, LoginDelegate);
}

void APicoSquarePlayerState::OnLoginComplete(bool bIsError, const FString& ErrorMessage, UPico_User* User)
{
    LoginDelegate.Clear();
    if (bIsError)
    {   
        PlatformLog(FString((TEXT("OnLoginComplete return Failed ErrorMessage : %s"), *ErrorMessage)));
        return;
    }
    PlatformLog(FString((TEXT("OnLoginComplete return successed! User Displayname : %s"), *User->GetDisplayName())));
    CurrentUser = User;
    bLoginComplete = true;
    APlatformSquareGameModeBase* GM = Cast<APlatformSquareGameModeBase>(UGameplayStatics::GetGameMode(this));
    if (GM)
    {
        GM->SetPlayerID(CurrentUser->GetID());
    }
    CheckUserDataAlready();

}

void APicoSquarePlayerState::GetLoginedUserFriendsAndRooms()
{
    PlatformLog(FString(TEXT("Call PicoSquarePlayerState::GetLoginedUserFriendsAndRooms!")));
    UserAndRoomList.Empty();
    LoginedInUserFriendsAndRoomsDelegate.Clear();
    LoginedInUserFriendsAndRoomsDelegate.BindDynamic(this, &APicoSquarePlayerState::OnGetUserFriendsAndRoomsComplete);
    UOnlinePicoUserFunction::PicoGetLoggedInUserAndRoom(this, LoginedInUserFriendsAndRoomsDelegate);
}

void APicoSquarePlayerState::OnGetUserFriendsAndRoomsComplete(bool bIsError, const FString& ErrorMessage, UPico_UserAndRoomArray* UserAndRoomArray)
{
    LoginedInUserFriendsAndRoomsDelegate.Clear();
    NextUserAndRoomArrayPage.Clear();
    if (bIsError)
    {
        PlatformLog(FString((TEXT("OnGetUserFriendsAndRoomsComplete return failed : %s"), *ErrorMessage)));
        return;
    }
    PlatformLog(FString(TEXT("OnGetUserFriendsAndRoomsComplete return successed!")));
    for (int32 i = 0; i < UserAndRoomArray->GetSize(); i++)
    {
        UserAndRoomList.Add(UserAndRoomArray->GetElement(i));
    }
    if (UserAndRoomArray->HasNextPage())
    {
        NextUserAndRoomArrayPage.BindDynamic(this, &APicoSquarePlayerState::OnGetUserFriendsAndRoomsComplete);
        UOnlinePicoUserFunction::PicoGetNextUserAndRoomArrayPage(this, UserAndRoomArray, NextUserAndRoomArrayPage);
    }
    else
    {
        bReadFriendsAndRoomsComplete = true;
        CheckUserDataAlready();
    }
}

void APicoSquarePlayerState::CheckUserDataAlready()
{
    if (bLoginComplete && bReadFriendsAndRoomsComplete)
    {
        bBaseDataAlready = true;
        PicoDataReadyDelegate.ExecuteIfBound();
        PicoDataReadyDelegate.Unbind();
    }
}

UFUNCTION(BlueprintCallable)
void RefreshUserData();

void APicoSquarePlayerState::RefreshUserData()
{
    bBaseDataAlready = false;
    bLoginComplete = false;
    bReadFriendsAndRoomsComplete = false;
    Login();
    GetLoginedUserFriendsAndRooms();

    PicoDataReadyDelegate.Unbind();
    PicoDataReadyDelegate.BindUObject(this, &APicoSquarePlayerState::CallUserDataAlready);
}

class PLATFORMSQUARE_API UVirtualSynchronyManager : public UObject
{}

// Convert binary data to and from structured data
void SaveLoadData(FArchive& Ar, FSyncInfo& SyncInfo);

// Read synchronized data within a room
bool ReadSyncInfo(FString& SendUserID, FSyncInfo& InSyncInfo);

// Send data to be synchronized within a room
bool SendSyncInfo(FSyncInfo& InSyncInfo);

bool UVirtualSynchronyManager::ReadSyncInfo(FString& SendUserID, FSyncInfo& InSyncInfo)
{
    UE_LOG(LogPICO, Log, TEXT("%s"), *PICO_FUNC_LINE);
    TArray<uint8> BinaryArray;
    bool ReadResult = UOnlinePicoNetworkingFunction::ReadPacket(this, BinaryArray, SendUserID);
    if (!ReadResult || BinaryArray.Num() == 0)
    {
        return false;
    }
    FMemoryReader FromBinary = FMemoryReader(BinaryArray, true);
    FromBinary.Seek(0);
    SaveLoadData(FromBinary, InSyncInfo);
    FromBinary.FlushCache();
    BinaryArray.Empty();
    FromBinary.Close();
    return true;
}

bool UVirtualSynchronyManager::SendSyncInfo(FSyncInfo& InSyncInfo)
{
    UE_LOG(LogPICO, Log, TEXT("%s"), *PICO_FUNC_LINE);
    FBufferArchive ToBinary;
    SaveLoadData(ToBinary, InSyncInfo);
    if (ToBinary.Num() <=0)
    {
        return false;
    }
    bool SendResult = UOnlinePicoNetworkingFunction::SendPacketToCurrentRoom(this, ToBinary, false);
    if (!SendResult)
    {
        UE_LOG(LogPICO, Log, TEXT("%s: Send sync info failed"), *PICO_FUNC_LINE);
    }
    ToBinary.FlushCache();
    ToBinary.Empty();
    ToBinary.Close();
    return SendResult;
}

class PLATFORMSQUARE_API UPICOGameInstanceSubsystem : public UGameInstanceSubsystem
{}

void OnJoinIntentChanged(const FString& DeeplinkMessage, const FString& DestinationApiName, const FString& LobbySessionId, const FString& MatchSessionId, const FString& Extra);

void UPICOGameInstanceSubsystem::Initialize(FSubsystemCollectionBase& Collection)
{
    Super::Initialize(Collection);
    JsonManager = NewObject<UJsonManager>();
    if (IOnlineSubsystem* Subsystem = IOnlineSubsystem::Get())
    {
        FOnlineSubsystemPico* PicoSubsystem = static_cast<FOnlineSubsystemPico*>(Subsystem);
        if (PicoSubsystem && PicoSubsystem->Init())
        {
            PicoPresenceInterface = PicoSubsystem->GetPicoPresenceInterface();
            if (PicoPresenceInterface)
            {
                PicoPresenceInterface->JoinIntentReceivedCallback.AddUObject(this, &UPICOGameInstanceSubsystem::OnJoinIntentChanged);
            }
        }
    }
}

void UPICOGameInstanceSubsystem::OnJoinIntentChanged(const FString& DeeplinkMessage, const FString& DestinationApiName, const FString& LobbySessionId, const FString& MatchSessionId, const FString& Extra)
{
    UE_LOG(LogPICO, Log, TEXT("%s, DeeplinkMessage: %s, DestinationApiName: %s, LobbySessionId: %s, MatchSessionId: %s, Extra: %s.!"), *PICO_FUNC_LINE, *DeeplinkMessage, *DestinationApiName, *LobbySessionId, *MatchSessionId, *Extra);
    bReceiveLaunchIntentChange = true;
    SaveDestinationApiName = DestinationApiName;
    SaveMatchSessionID = MatchSessionId;
}

UFUNCTION(BlueprintCallable)
bool GetIntentChanged(FString& OutDestinationApiName, FString& OutMatchSessionId);

bool UPICOGameInstanceSubsystem::GetIntentChanged(FString& OutDestinationApiName, FString& OutMatchSessionId)
{
    if (bReceiveLaunchIntentChange)
    {
        OutDestinationApiName = SaveDestinationApiName;
        OutMatchSessionId = SaveMatchSessionID;
        bReceiveLaunchIntentChange = false;
        SaveDestinationApiName = FString();
        SaveMatchSessionID = FString();
        return true;
    }
    return false;
}
