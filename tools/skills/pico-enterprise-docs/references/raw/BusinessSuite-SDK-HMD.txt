}

public void sendMsg(String msg) {

//注册接收消息
public void setMsgReceiver(ICmdReceiver receiver) {

import android.app.Activity;
import android.os.Bundle;
import android.os.IBinder;
import android.text.method.ScrollingMovementMethod;
import android.util.Log;
import android.widget.Button;
import android.widget.EditText;
import android.widget.TextView;
import com.picoxr.clientservice.common.server.model.RevCmdModel;
import com.picoxr.clientservice.service.ICmdReceiver;


public class MainActivity extends Activity {
    private static final String TAG = "EASDKDemo MainActivity";
    private String results = "";
    @Override
    protected void onCreate(Bundle savedInstanceState) {
        super.onCreate(savedInstanceState);
        Log.i(TAG, "onCreate");
        setContentView(R.layout.activity_main);

        Button bindButton = findViewById(R.id.bind_btn);
        bindButton.setClickable(true);
        bindButton.setOnClickListener(v -> {
            Log.i(TAG, "click bindButton");
            EASDKManager.getInstance(this).bindService();
        });

        Button unbindButton = findViewById(R.id.unbind_btn);
        unbindButton.setClickable(true);
        unbindButton.setOnClickListener(v -> {
            Log.i(TAG, "click unbindButton");
            EASDKManager.getInstance(this).unbindService();
        });

        EditText cmdParamsET = findViewById(R.id.send_controller_cmd_params);
        Button sendControllerButton = findViewById(R.id.send_controller_cmd);
        sendControllerButton.setClickable(true);
        sendControllerButton.setOnClickListener(v -> {
            Log.i(TAG, "click send to controller Button");
            String params = cmdParamsET.getText().toString();
            EASDKManager.getInstance(this).sendMsg(params);
        });

        TextView receiveControllerTV = findViewById(R.id.receive_controller_msg);
        Button setReceiverButton = findViewById(R.id.set_controller_receiver);
        setReceiverButton.setClickable(true);
        setReceiverButton.setOnClickListener(v -> {
            Log.i(TAG, "click setReceiverButton");
            ICmdReceiver receiver = new ICmdReceiver.Stub() {
                @Override
                public void receiveData(RevCmdModel revCmdModel) {
                    if (revCmdModel != null) {
                        String msg = revCmdModel.getParamsContent();
                        if (!"".equals(results)) {
                            results = msg + "\n";
                        }
                        results = results + msg;
                        runOnUiThread(() -> receiveControllerTV.setText(results));
                    }
                }

                @Override
                public IBinder asBinder() {
                    return this;
                }
            };
            EASDKManager.getInstance(this).setMsgReceiver(receiver);
        });
    }
}
