 void pbsStartRecordAudio(in AudioFormat audioFormat, in IBytesCallback callback, int ext);
参数名称
类型
说明
audioFormat
AudioFormat
详见：android.media.AudioFormat
callback
IBytesCallback
音频流回调接口
ext
int 
扩展位，无意义
AudioFormat audioFormat = new AudioFormat.Builder()
        .setChannelMask(AudioFormat.CHANNEL_IN_MONO)
        .setEncoding(AudioFormat.ENCODING_PCM_16BIT)
        .setSampleRate(44100)
        .build();

IBytesCallback.Stub callback = new IBytesCallback.Stub(){
    @Override
    public void callBack(byte[] buffer, int bufferSize) throws RemoteException {
        Log.i(TAG, "callBack: " + bufferSize + "  " + Arrays.toString(buffer));
        try {
            os.write(buffer);
        } catch (IOException e) {
            e.printStackTrace();
        }
    }
};
try {
    ToBServiceHelper.getInstance().getServiceBinder().pbsStartRecordAudio(audioFormat,callback,0);
} catch (RemoteException e) {
    e.printStackTrace();
}
