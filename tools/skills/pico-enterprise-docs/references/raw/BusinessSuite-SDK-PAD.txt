向VR头戴发送任何指令之前，必须先锁定设备；如果只有一个控制端，可以在事件回调的onServerConnect()和onDeviceOnline(String devUid)中调用锁定设备接口，反之则需要根据业务场景分别调用锁定设备和解除锁定设备接口。

private ArrayList<String> onlineDevices = new ArrayList<>();
private EventListener eventListener = new EventListener(){
    @Override
    public void onServerConnect() {
        Log.i(TAG, "onServerConnect");
        //服务连接或重连成功，遍历锁定在线设备
        for (String devUid : mDeviceIds){
            try {
                PbsBroadcastingHelper.getInstance().getServiceBinder().lockDevice(devUid);
            } catch (RemoteException e) {
                Log.e(TAG,"lockDevice error",e);
            }            
        }
    }
    
    @Override
    public void onServerDisconnect() {
        Log.i(TAG, "onServerDisconnect");
    }
    
    @Override
    public void onDeviceOnline(String devUid) {
        Log.i(TAG, "onDeviceOnline: "+devUid);
        if (!onlineDevices.contains(devUid)) {
            onlineDevices.add(devUid);
            //设备上线锁定设备
            try {
                PbsBroadcastingHelper.getInstance().getServiceBinder().lockDevice(devUid);
            } catch (RemoteException e) {
                Log.e(TAG,"lockDevice error",e);
            }
        }
    }
    
    @Override
    public void onDeviceOffline(String devUid) {
        Log.i(TAG, "onVRDeviceOffline: "+ devUid);
        if (onlineDevices.contains(devUid)) {
            onlineDevices.remove(devUid);
        }
    }
    
    ...
}
