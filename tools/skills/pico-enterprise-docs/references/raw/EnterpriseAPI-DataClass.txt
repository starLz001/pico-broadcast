public class LargeSpaceQuickModeInfo {
    /**
     * 快捷模式开关状态
     */
    public boolean status;
    
    /**
    * 安全区长度
    */
    public int length;
    /**
    * 安全区宽度
    */
    public int width;

    /**
     * 设定原点方式
     * -1 - 未设定， 0 - 默认,1 - 扫描mark
     */
    public int originType;

    @Override
    public String toString() {
        return "LargeSpaceShortcutModeInfo{" +
                "status=" + status +
                ", length=" + length +
                ", width=" + width +
                ", originType=" + originType +
                '}';
    }
}

public class Pose{

    public long timestamp;  // nanoseconds
    public double x;           // positon X
    public double y;           // position Y
    public double z;           // position Z
    public double rw;          // rotation W
    public double rx;          // rotation X
    public double ry;          // rotation Y
    public double rz;          // rotation Z
    public int type;       // 1:6DOF 0:3DOF
    public int confidence; // 1:good 0:bad
    public int poseError;

    List<Integer> reservedInt;
    List<Double> reservedDouble;
    
    public static class PoseErrorType {
        int BRIGHT_LIGHT_ERROR = (1);
        int LOW_LIGHT_ERROR = (1 << 1);
        int LOW_FEATURE_COUNT_ERROR = (1 << 2);
        int CAMERA_CALIBRATION_ERROR = (1 << 3);
        int RELOCATION_IN_PROGRESS = (1 << 4);
        int INITILIZATION_IN_PROGRESS = (1 << 5);
        int NO_CAMERA_ERROR = (1 << 6);
        int NO_IMU_ERROR = (1 << 7);
        int IMU_JITTER_ERROR = (1 << 8);
        int UNKNOWN_ERROR = (1 << 9);
    }
}
